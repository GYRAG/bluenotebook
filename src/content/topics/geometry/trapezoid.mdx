<Definition>ტრაპეცია ეწოდება ოთხკუთხედს, რომლის ორი მოპირდაპირე გვერდი პარალელურია, ხოლო დანარჩენი ორი — არა. პარალელურ გვერდებს ($AD \parallel BC$) **ფუძეები** ეწოდება, დანარჩენ ორს — **ფერდები**. ფერდების შუაწერტილების შემაერთებელი მონაკვეთი ტრაპეციის **შუახაზია**.</Definition>

<Remark kind="convention">ქართულ სკოლაში ტრაპეციას ზუსტად ერთი წყვილი პარალელური გვერდი აქვს, ამიტომ პარალელოგრამი ტრაპეცია არ არის. ზოგ სახელმძღვანელოში (განსაკუთრებით ინგლისურენოვანში) ტრაპეციად ითვლება ყველა ოთხკუთხედი, რომელსაც *ერთი მაინც* წყვილი პარალელური გვერდი აქვს — მაშინ პარალელოგრამიც ტრაპეციაა.</Remark>

<Properties>

<Property id="angles-on-legs" title="ერთ ფერდთან მდებარე კუთხეების ჯამი $180^\circ$-ია" why="ეს შიგა ცალმხრივი კუთხეებია პარალელურ ფუძეებთან.">
<Proof given="$ABCD$ — ტრაპეცია, $AD \parallel BC$" prove="$\angle A + \angle B = 180^\circ,\ \angle C + \angle D = 180^\circ$">
<Step hl="AD||BC AB">$AD \parallel BC$, ხოლო ფერდი $AB$ მათი მკვეთია.</Step>
<Step hl="<A <B" tex="\angle A + \angle B = 180^\circ">$\angle A$ და $\angle B$ შიგა ცალმხრივი კუთხეებია.</Step>
<Step hl="<C <D" tex="\angle C + \angle D = 180^\circ \quad \blacksquare">იგივე მეორე ფერდისთვის ($CD$ მკვეთი).</Step>
</Proof>
</Property>

<Property id="midline" title="შუახაზი ფუძეების პარალელურია და მათი ჯამის ნახევრის ტოლია" why="ტრაპეცია სამკუთხედად „გადაკეთდება“, რომლის შუახაზიც იგივეა.">
<Proof given="$M, N$ — $AB$ და $CD$ ფერდების შუაწერტილები" prove="$MN \parallel AD,\ MN = \tfrac{a + b}{2}$">
<Step show="M N" hl="M N">$M$ და $N$ ფერდების შუაწერტილებია.</Step>
<Step show="BE DE E" hl="BE">გავავლოთ $BN$ და გავაგრძელოთ, სანამ $AD$ წრფეს $E$ წერტილში არ გადაკვეთს.</Step>
<Step hl="BCN EDN CN=DN" tex="\triangle BCN = \triangle EDN">$CN = DN$, $\angle BNC = \angle END$ (ვერტიკალური), $\angle BCN = \angle EDN$ (ჯვარედინი, $BC \parallel AE$) — სამკუთხედები ტოლია (კგკ).</Step>
<Step hl="BN=NE BC=DE" tex="BN = NE, \qquad DE = BC = b">ტოლი სამკუთხედების შესაბამისი გვერდები.</Step>
<Step show="MN" hl="MN ABE" tex="MN = \tfrac12 AE = \tfrac{a + b}{2} \quad \blacksquare">$MN$ შეაერთებს $\triangle ABE$-ის $AB$ და $BE$ გვერდების შუაწერტილებს — ის ამ სამკუთხედის შუახაზია; $AE = AD + DE = a + b$.</Step>
</Proof>
</Property>

<Property id="area" title="ტრაპეციის ფართობი ფუძეების ჯამის ნახევრისა და სიმაღლის ნამრავლია" why="დიაგონალი მას ორ სამკუთხედად ყოფს ერთი და იმავე სიმაღლით.">
<Proof given="$AD = a,\ BC = b$, სიმაღლე $h$" prove="$S = \tfrac{a + b}{2}\, h$">
<Step show="BD BH <BHD" hl="BD">გავავლოთ დიაგონალი $BD$ და სიმაღლე $BH$.</Step>
<Step hl="ABD" tex="S_{ABD} = \tfrac12\, a\,h">$\triangle ABD$-ის ფუძეა $AD = a$, სიმაღლე — $h$.</Step>
<Step hl="BCD" tex="S_{BCD} = \tfrac12\, b\,h">$\triangle BCD$-ის ფუძეა $BC = b$; $D$-დან $BC$ წრფემდე მანძილიც $h$-ია (პარალელურ წრფეებს შორის მანძილი).</Step>
<Step tex="S = \tfrac12 a h + \tfrac12 b h = \frac{a + b}{2}\, h = m\,h \quad \blacksquare">შევკრიბოთ.</Step>
</Proof>
</Property>

<Property id="diagonals-ratio" title="დიაგონალების გადაკვეთის წერტილი თითოეულ დიაგონალს ფუძეების შეფარდებით ყოფს" why="ფუძეებზე აგებული ორი სამკუთხედი მსგავსია.">
<Proof given="$O = AC \cap BD$" prove="$AO : OC = DO : OB = a : b$">
<Step show="AC BD O" hl="AC BD">გავავლოთ დიაგონალები.</Step>
<Step hl="<AOD=<COB <OAD=<OCB">$\angle AOD = \angle COB$ (ვერტიკალური), $\angle OAD = \angle OCB$ (ჯვარედინი, $AD \parallel BC$).</Step>
<Step hl="AOD BOC" tex="\triangle AOD \sim \triangle COB \;\Rightarrow\; \frac{AO}{OC} = \frac{DO}{OB} = \frac{AD}{CB} = \frac{a}{b} \quad \blacksquare">სამკუთხედები მსგავსია ორი კუთხით.</Step>
</Proof>
</Property>

<Property id="segment-through-o" title="დიაგონალების გადაკვეთაზე გამავალი, ფუძეების პარალელური მონაკვეთი $\tfrac{2ab}{a+b}$-ია და $O$ მას შუაზე ყოფს" why="ორი წყვილი მსგავსი სამკუთხედი ერთსა და იმავე $\tfrac{ab}{a+b}$-ს იძლევა.">
<Proof given="$PQ \parallel AD$, $O \in PQ$, $P \in AB$, $Q \in CD$" prove="$PQ = \tfrac{2ab}{a + b}$">
<Step show="AC BD O PQ P Q" hl="PQ" tex="\frac{AO}{AC} = \frac{a}{a + b}">წინა თვისებიდან $AO : OC = a : b$.</Step>
<Step hl="APO ABC" tex="\frac{PO}{BC} = \frac{AO}{AC} \;\Rightarrow\; PO = \frac{ab}{a + b}">$PO \parallel BC$, ამიტომ $\triangle APO \sim \triangle ABC$.</Step>
<Step hl="DQO DCB" tex="\frac{OQ}{BC} = \frac{DO}{DB} = \frac{a}{a + b} \;\Rightarrow\; OQ = \frac{ab}{a + b}">$OQ \parallel BC$, ამიტომ $\triangle DQO \sim \triangle DCB$.</Step>
<Step hl="PO=OQ" tex="PQ = PO + OQ = \frac{2ab}{a + b} \quad \blacksquare">$PO = OQ$ — $O$ ამ მონაკვეთის შუაწერტილია.</Step>
</Proof>
</Property>

<Property id="diagonal-midpoints" title="დიაგონალების შუაწერტილები შუახაზზე ძევს და მათ შორის მანძილი ფუძეების სხვაობის ნახევარია ($a > b$)" why="$M$-ზე ორი სამკუთხედის შუახაზი გადის: $\tfrac{b}{2}$ და $\tfrac{a}{2}$.">
<Proof given="$X, Y$ — $AC$ და $BD$ დიაგონალების შუაწერტილები, $a > b$" prove="$X, Y \in MN,\ XY = \tfrac{a - b}{2}$">
<Step show="AC BD X Y M" hl="X Y">$X$ და $Y$ დიაგონალების შუაწერტილებია, $M$ — $AB$ ფერდის შუაწერტილი.</Step>
<Step show="MX" hl="MX ABC" tex="MX \parallel BC, \qquad MX = \tfrac{b}{2}">$MX$ აერთებს $\triangle ABC$-ის ორი გვერდის შუაწერტილებს — ის ამ სამკუთხედის შუახაზია.</Step>
<Step show="MY" hl="MY ABD" tex="MY \parallel AD, \qquad MY = \tfrac{a}{2}">$MY$ — $\triangle ABD$-ის შუახაზი.</Step>
<Step show="MN N" hl="MN">$MX$ და $MY$ $M$-ზე გადის და ფუძეების პარალელურია. $M$-ზე ფუძეების პარალელური მხოლოდ ერთი წრფე გადის — შუახაზის წრფე, ამიტომ $X$ და $Y$ მასზე ძევს.</Step>
<Step hl="XY" tex="XY = MY - MX = \frac{a - b}{2} \quad \blacksquare">$X$ და $Y$ $M$-დან ერთ მხარესაა და $MX < MY$.</Step>
</Proof>
</Property>

<Property id="diagonal-triangles" title="დიაგონალები ტრაპეციას ოთხ სამკუთხედად ყოფს: ფერდებთან მდებარე სამკუთხედები ტოლდიდია, ხოლო თითოეულის ფართობის კვადრატი ფუძეებთან მდებარე სამკუთხედების ფართობების ნამრავლია" why="ერთი და იმავე ფუძისა და სიმაღლის სამკუთხედები ტოლდიდია.">
<Proof given="$O = AC \cap BD$" prove="$S_{AOB} = S_{COD},\ S_{AOB}^2 = S_{AOD} \cdot S_{BOC}$">
<Step show="AC BD O" hl="ABD ACD" tex="S_{ABD} = S_{ACD}">ორ სამკუთხედს საერთო ფუძე $AD$ აქვს, სიმაღლეები კი ტოლია: $B$ და $C$ ერთნაირად არის დაშორებული $AD$-დან.</Step>
<Step hl="AOB COD" tex="S_{AOB} = S_{ABD} - S_{AOD} = S_{ACD} - S_{AOD} = S_{COD}">ორივეს გამოვაკლოთ $S_{AOD}$.</Step>
<Step hl="AOB AOD" tex="\frac{S_{AOB}}{S_{AOD}} = \frac{OB}{OD}">სამკუთხედებს საერთო წვერო $A$ აქვს, ფუძეები $OB$ და $OD$ კი ერთ წრფეზე ძევს — ფართობები ფუძეების პროპორციულია.</Step>
<Step hl="BOC COD" tex="\frac{S_{BOC}}{S_{COD}} = \frac{OB}{OD}">იგივე, საერთო წვეროთი $C$.</Step>
<Step tex="S_{AOB} \cdot S_{COD} = S_{AOD} \cdot S_{BOC} \;\Rightarrow\; S_{AOB}^2 = S_{AOD} \cdot S_{BOC} \quad \blacksquare">ორი შეფარდება ტოლია; ჩავსვათ $S_{COD} = S_{AOB}$.</Step>
</Proof>
</Property>

<Property id="leg-bisectors" title="ერთ ფერდთან მდებარე კუთხეების ბისექტრისები მართი კუთხით იკვეთება, ხოლო გადაკვეთის წერტილი შუახაზის წრფეზე ძევს" why="ნახევრების ჯამი $90^\circ$-ია; ჰიპოტენუზაზე გავლებული მედიანა მისი ნახევარია.">
<Proof given="$AL$, $BL$ — $\angle A$-სა და $\angle B$-ს ბისექტრისები" prove="$\angle ALB = 90^\circ$, $L \in MN$">
<Step show="AL BL L" hl="AL BL">$AL$ და $BL$ — $AB$ ფერდთან მდებარე კუთხეების ბისექტრისები.</Step>
<Step hl="<ALB" tex="\angle LAB + \angle LBA = \tfrac12(\angle A + \angle B) = 90^\circ">ერთ ფერდთან მდებარე კუთხეების ჯამი $180^\circ$-ია, ამიტომ $\triangle ALB$-ში $\angle ALB = 90^\circ$.</Step>
<Step show="ML M" hl="ML=AM" tex="ML = \tfrac12 AB = AM">$LM$ მართკუთხა $\triangle ALB$-ის მედიანაა ჰიპოტენუზაზე.</Step>
<Step hl="<MLA=<LAD" tex="\angle MLA = \angle MAL = \angle LAD">$\triangle AML$ ტოლფერდაა, $AL$ კი ბისექტრისაა.</Step>
<Step show="MN N" hl="ML||AD">$\angle MLA$ და $\angle LAD$ ჯვარედინი კუთხეებია, ამიტომ $ML \parallel AD$. $M$-ზე $AD$-ს პარალელური მხოლოდ შუახაზის წრფე გადის — $L$ მასზე ძევს. ∎</Step>
</Proof>
</Property>

</Properties>

<Remark>$\tfrac{2ab}{a+b}$ არის $a$ და $b$ რიცხვების საშუალო ჰარმონიული. ის ყოველთვის შუახაზზე ($\tfrac{a+b}{2}$) ნაკლებია, თუ $a \ne b$.</Remark>

<Property id="isosceles-base-angles" title="ტოლფერდა ტრაპეციის კუთხეები თითოეულ ფუძესთან ტოლია" why="ორი სიმაღლე ორ ტოლ მართკუთხა სამკუთხედს ჩამოჭრის.">
<Proof given="$AB = CD$" prove="$\angle A = \angle D,\ \angle B = \angle C$">
<Step set={{ a: 6, b: 3, s: 1.5 }} show="BH CK <BHA <CKD" hl="BH CK">დავუშვათ სიმაღლეები $BH$ და $CK$; $BH = CK$ (პარალელურ ფუძეებს შორის მანძილი).</Step>
<Step hl="ABH DCK AB=CD BH=CK" tex="\triangle ABH = \triangle DCK">მართკუთხა სამკუთხედები ტოლია ჰიპოტენუზისა ($AB = DC$) და კათეტის ($BH = CK$) მიხედვით.</Step>
<Step hl="<A=<D" tex="\angle A = \angle D">შესაბამისი კუთხეები ტოლია.</Step>
<Step hl="<B=<C" tex="\angle B = 180^\circ - \angle A = 180^\circ - \angle D = \angle C \quad \blacksquare">ერთ ფერდთან მდებარე კუთხეების ჯამი $180^\circ$-ია.</Step>
</Proof>
</Property>

<Property id="isosceles-diagonals" title="ტოლფერდა ტრაპეციის დიაგონალები ტოლია" why="ორი სამკუთხედი საერთო ფუძით და ტოლი ფერდებით.">
<Proof given="$AB = CD$" prove="$AC = BD$">
<Step set={{ a: 6, b: 3, s: 1.5 }} show="AC BD" hl="AC BD">გავავლოთ დიაგონალები.</Step>
<Step hl="ABD AB=DC <A=<D" tex="AB = DC,\quad AD = DA,\quad \angle A = \angle D">განვიხილოთ $\triangle ABD$ და $\triangle DCA$: ფერდები ტოლია, $AD$ საერთოა, ფუძესთან კუთხეები ტოლია (წინა თვისება).</Step>
<Step hl="AC=BD" tex="\triangle ABD = \triangle DCA \;\Rightarrow\; BD = CA \quad \blacksquare">გკგ.</Step>
</Proof>
</Property>

<Property id="isosceles-projections" title="ტოლფერდა ტრაპეციაში სიმაღლე დიდ ფუძეს $\tfrac{a-b}{2}$ და $\tfrac{a+b}{2}$ მონაკვეთებად ყოფს, დიაგონალი კი $d^2 = h^2 + \left(\tfrac{a+b}{2}\right)^2$" why="ორი სიმაღლე ორ ტოლ მართკუთხა სამკუთხედს ჩამოჭრის, შუაში კი მართკუთხედი რჩება.">
<Proof given="$AB = CD$, $BH$ და $CK$ — სიმაღლეები" prove="$AH = \tfrac{a-b}{2},\ HD = \tfrac{a+b}{2}$">
<Step set={{ a: 6, b: 3, s: 1.5 }} show="BH CK <BHA <CKD" hl="BH CK">დავუშვათ სიმაღლეები $BH$ და $CK$.</Step>
<Step hl="AH KD" tex="AH = KD">$\triangle ABH = \triangle DCK$ (ჰიპოტენუზით და კათეტით).</Step>
<Step hl="HK" tex="HK = BC = b">$HBCK$ მართკუთხედია.</Step>
<Step hl="AH HD" tex="AH = \frac{a - b}{2}, \qquad HD = HK + KD = \frac{a + b}{2}">$AD = AH + HK + KD = 2\,AH + b = a$.</Step>
<Step show="BD" hl="BHD" tex="d^2 = BD^2 = h^2 + \left(\frac{a + b}{2}\right)^2 \quad \blacksquare">პითაგორას თეორემა მართკუთხა $\triangle BHD$-ში.</Step>
</Proof>
</Property>

<Property id="isosceles-perpendicular-diagonals" title="თუ ტოლფერდა ტრაპეციის დიაგონალები ურთიერთმართობია, მისი სიმაღლე შუახაზის ტოლია: $h = \tfrac{a+b}{2}$, $S = h^2$" why="დიაგონალი ფუძესთან $45^\circ$-ს ადგენს, ამიტომ სიმაღლე ფუძის თავის მონაკვეთს უდრის.">
<Proof given="$AB = CD$, $AC \perp BD$" prove="$h = \tfrac{a+b}{2},\ S = h^2$">
<Step set={{ a: 4, b: 2, h: 3, s: 1 }} show="AC BD O" hl="<AOD">დიაგონალები $O$-ში მართი კუთხით იკვეთება.</Step>
<Step hl="AOD" tex="\angle OAD = \angle ODA = 45^\circ">$\triangle ABD = \triangle DCA$ (იხ. დიაგონალების ტოლობა), ამიტომ $\angle ODA = \angle OAD$; მათი ჯამი $90^\circ$-ია.</Step>
<Step show="BH <BHD" hl="BHD" tex="h = BH = HD = \frac{a + b}{2}">მართკუთხა $\triangle BHD$-ში $\angle BDH = 45^\circ$, ამიტომ ის ტოლფერდაა; $HD = \tfrac{a+b}{2}$ წინა თვისებით.</Step>
<Step tex="S = \frac{a + b}{2} \cdot h = h \cdot h = h^2 \quad \blacksquare">ფართობის ფორმულა.</Step>
</Proof>
</Property>

<Property id="isosceles-cyclic" title="ტოლფერდა ტრაპეციის გარშემო წრეწირის შემოხაზვა შეიძლება" why="მოპირდაპირე კუთხეების ჯამი $180^\circ$ გამოდის.">

<Step hl="ABCD">მოპირდაპირე კუთხეების ჯამი $180^\circ$-ია — ოთხკუთხედის გარშემო წრეწირის შემოხაზვა შეიძლება. ∎</Step>
</Proof>

<Property id="criterion-base-angles" kind="converse" title="თუ ტრაპეციის კუთხეები ფუძესთან ტოლია, ის ტოლფერდაა" why="იგივე ორი მართკუთხა სამკუთხედი, ოღონდ კათეტითა და კუთხით.">

<Property id="criterion-diagonals" kind="converse" title="თუ ტრაპეციის დიაგონალები ტოლია, ის ტოლფერდაა" why="ერთი დიაგონალის პარალელური გადატანა ტოლფერდა სამკუთხედს ქმნის.">
